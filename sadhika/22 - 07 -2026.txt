class Solution:
    def removeDuplicates(self, nums):
        if  not nums:
            return 0
        new =[nums[0]]  #creating a new array with first value is 0
        for i in range(1,len(nums)):
            if nums[i]!=nums[i-1]:
                new.append(nums[i])
        for i in range(len(new)):
            nums[i]=new[i]
        return len(new)        

output=[1,2]

class Solution:
    def removeDuplicates(self, nums):
        #using sets
        unique=sorted(set(nums))
        for i in range(len(unique)):
            nums[i]=unique[i]
        return len(unique)
